name: "Decrypt una cadena"
description: "Descifra usando una llave. Expone plaintext (no lo pases entre jobs)."

inputs:
  ciphertext:
    description: "Ciphertext en Base64 (una sola línea)"
    required: true
  host:
    description: "HOSTNAME usado en el cifrado (proporcionado por el job anterior)"
    required: true
  extra_key:
    description: "PARTE 2 de la llave usado para cifrar"
    required: true
  openssl_alg:
    description: "Algoritmo OpenSSL"
    required: false
    default: "aes-256-cbc"
  openssl_iter:
    description: "Iteraciones PBKDF2"
    required: false
    default: "200000"

outputs:
  PLAINTEXT:
    description: "Texto plano (descifrado). Úsalo SOLO en el mismo job."
    value: ${{ steps.decrypt.outputs.PLAINTEXT }}

runs:
  using: "composite"
  steps:
    - name: Decrypt (host + extra_key)
      id: decrypt
      shell: bash
      env:
        CIPHER: ${{ inputs.ciphertext }}
        HOST_FOR_KEY: ${{ inputs.host }}
        KEY_PART2: ${{ inputs.extra_key }}
        ALG: ${{ inputs.openssl_alg }}
        ITER: ${{ inputs.openssl_iter }}
      run: |
        set -euo pipefail
        KEY="${HOST_FOR_KEY}${KEY_PART2}"
        echo "::add-mask::${KEY}"

        # Descifrar (mismas opciones que en el cifrado)
        PT="$(printf '%s' "${CIPHER}" \
                      | openssl enc -"${ALG}" -pbkdf2 -iter "${ITER}" -md sha256 \
                                  -d -salt -base64 -A -pass pass:"${KEY}")"
        {
          echo "PLAINTEXT<<EOF"
          printf '%s\n' "${PT}"
          echo "EOF"
        } >> "${GITHUB_OUTPUT}"

name: "Encriptar una cadena"
description: "Cifra un texto usando una llave. Expone outputs: cipher, host."
inputs:
  plaintext:
    description: "Texto en claro a cifrar"
    required: true
  extra_key:
    description: "PARTE 2 de la llave usado para cifrar"
    required: true
  openssl_alg:
    description: "Algoritmo OpenSSL"
    required: false
    default: "aes-256-cbc"
  openssl_iter:
    description: "Iteraciones PBKDF2"
    required: false
    default: "200000"

outputs:
  CIPHER:
    description: "Ciphertext (Base64 una sola línea)"
    value: ${{ steps.encrypt.outputs.CIPHER }}
  HOST:
    description: "HOSTNAME usado en el cifrado, PARTE 1 de la llave"
    value: ${{ steps.encrypt.outputs.HOST }}

runs:
  using: "composite"
  steps:
    - name: Encrypt
      id: encrypt
      shell: bash
      env:
        PLAINTEXT: ${{ inputs.plaintext }}
        KEY_PART2: ${{ inputs.extra_key }}
        ALG: ${{ inputs.openssl_alg }}
        ITER: ${{ inputs.openssl_iter }}
      run: |
        set -euo pipefail
        HOST_EPHEMERAL="$(hostname)"
        KEY="${HOST_EPHEMERAL}${KEY_PART2}"
        echo "::add-mask::${PLAINTEXT}"
        echo "::add-mask::${KEY}"

        #Cifrar (OpenSSL AES-256-CBC + PBKDF2 + 200k iter + Base64 en una sola línea)
        #    *NO* imprime variables en consola; solo redirige a $GITHUB_OUTPUT
        CT="$(printf '%s' "${PLAINTEXT}" \
              | openssl enc -"${ALG}" -pbkdf2 -iter "${ITER}" -md sha256 \
                            -salt -base64 -A -pass pass:"${KEY}")"

        #Exponer outputs: ciphertext y hostname
        {
          echo "CIPHER=${CT}"
          echo "HOST=${HOST_EPHEMERAL}"
        } >> "${GITHUB_OUTPUT}"

            errors=0
            local dir="$1"
            if [[ ! -d "$dir" ]]; then
              return 0
            fi
            num_file=$(find "$dir" -maxdepth 1 -type f -name '*.yaml' | wc -l)
            if [[ "$num_file" -eq 0 ]]; then
              return 0
            fi
            for stmt in "$dir"/*.yaml; do
              local tmp="${stmt}.with.environment"
              sed "s/\${environment}/${ENVIRONMENT}/g" "$stmt" > "$tmp"
              sa=$(yq -r '."service-account" // ""' "$tmp")
              ak=$(yq -r '."api-key" // ""' "$tmp")
              #Validar
              if [[ -z "${sa}" || "${sa}" == "null" || "x${sa}" == "x"  ]]; then
                echo "❌ sa vacío o null en ${stmt}"
                errors=$((errors+1))
              fi
              if [[ -z "${ak}" || "${ak}" == "null" || "x${ak}" == "x"  ]]; then
                echo "❌ ak vacío o null en ${stmt}"

              fi

                errors=$((errors+1))
              fi
              if [[ ! "$ak" =~ $regex_ak ]]; then
                echo "❌ ak NO cumple formato: $ak. Esperado: $regex_ak"
                errors=$((errors+1))
              fi

                + "'"$PATH_FINAL"'" + $sa + "/" + $ak + " password | SECRET_" + $ak + " ;"
                ' "$tmp"
            done
            if [[ "$errors" -gt 0 ]]; then
              echo "❌ Validación falló con $errors error(es)."
              exit 1
            fi
          }

                + "'"$PATH_FINAL"'" + $sa + "/" + $ak + " password | SECRET_" + $ak + " ;"

#!/bin/bash
set -euo pipefail
export ENVIRONMENT_ID=$1
echo "ENVIRONMENT_ID: $ENVIRONMENT_ID"
export RELATIVE_SUBPATH_SRC=$2
echo "RELATIVE_SUBPATH_SRC=$RELATIVE_SUBPATH_SRC"
export RELATIVE_SUBPATH_TERRAFORM=$3
echo "RELATIVE_SUBPATH_TERRAFORM:$RELATIVE_SUBPATH_TERRAFORM"
AUTOMATION_DIRNAME="./resources"
echo "AUTOMATION_DIRNAME: $AUTOMATION_DIRNAME"
#
echo "ENVIRONMENT_HV: $ENVIRONMENT_HV"
#VARS_FILE="$ENVIRONMENT_HV-vars.yaml"
VARS_FILE="cc-compute-pools.yaml"
echo "VARS_FILE: $VARS_FILE"
#
YAML_FILE_CPS_FLINK_PATH="${RELATIVE_SUBPATH_SRC}/$VARS_FILE"
echo "YAML_FILE_CPS_FLINK_PATH: $YAML_FILE_CPS_FLINK_PATH"
#
#Valida número de yamls
if [[ ! -f "$YAML_FILE_CPS_FLINK_PATH" ]]; then
  echo "REVISAR: Archivo no existe $YAML_FILE_CPS_FLINK_PATH"
  exit 0
fi
#
ARCHIVO_TPL_CP_FLINK="${AUTOMATION_DIRNAME}/template/cp_flink.tf.tpl"
ARCHIVO_TF_CP_FLINK=$(basename "${ARCHIVO_TPL_CP_FLINK%.*}")
rm -f $ARCHIVO_TF_CP_FLINK
touch $ARCHIVO_TF_CP_FLINK
#
yq -r '.compute_pools[] | [.cloud,.region,.max_cfu,.pool_name] | @tsv' $YAML_FILE_CPS_FLINK_PATH \
| while IFS=$'\t' read -r cloud region max_cfu display_name; do
  export CLOUD="${cloud}"
  export REGION="${region}"  
  export MAX_CFU="${max_cfu}" 
  export DISPLAY_NAME="${display_name}"
  echo "$CLOUD $REGION $MAX_CFU $DISPLAY_NAME"
  cat <<EOF >> "$ARCHIVO_TF_CP_FLINK"
$(envsubst < $ARCHIVO_TPL_CP_FLINK)
EOF
done
#
mv $ARCHIVO_TF_CP_FLINK $RELATIVE_SUBPATH_TERRAFORM
echo ""
echo "********** contenido de archivo: $ARCHIVO_TF_CP_FLINK *******"
cat $RELATIVE_SUBPATH_TERRAFORM/$ARCHIVO_TF_CP_FLINK
echo "FIN - LOAD STATEMENTS, CPS"
